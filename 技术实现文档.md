### 4.2 Router 实现

```javascript
// router.js
class Router {
  constructor() {
    this.routes = new Map();
    this.currentRoute = null;
    this.history = [];

    window.addEventListener('hashchange', () => this.handleRoute());
    window.addEventListener('load', () => this.handleRoute());
  }

  register(pattern, handler) {
    this.routes.set(pattern, handler);
  }

  navigate(hash) {
    window.location.hash = hash;
  }

  back() {
    if (this.history.length > 1) {
      this.history.pop();
      window.location.hash = this.history[this.history.length - 1];
    } else {
      window.location.hash = '#/';
    }
  }

  // 限制历史栈最大长度，防止内存泄漏
  _pushHistory(hash) {
    this.history.push(hash);
    if (this.history.length > 50) {
      this.history = this.history.slice(-50);
    }
  }

  handleRoute() {
    const hash = window.location.hash || '#/';

    // 防止 404 死循环：如果当前已经在首页，不再重定向
    if (this._isNavigating404) return;
    this._isNavigating404 = true;

    // 匹配路由
    for (const [pattern, handler] of this.routes) {
      const match = this.matchPattern(pattern, hash);
      if (match) {
        this._isNavigating404 = false;
        // 页面切换动画
        this.animateTransition(() => {
          handler(match.params);
          this.currentRoute = hash;
          this._pushHistory(hash);
        });
        return;
      }
    }

    // 404 回首页
    this._isNavigating404 = false;
    if (hash !== '#/') {
      this.navigate('#/');
    }
  }

  matchPattern(pattern, hash) {
    // 支持 :id 参数匹配
    const patternParts = pattern.split('/');
    const hashParts = hash.split('/');

    if (patternParts.length !== hashParts.length) return null;

    const params = {};
    for (let i = 0; i < patternParts.length; i++) {
      if (patternParts[i].startsWith(':')) {
        params[patternParts[i].slice(1)] = hashParts[i];
      } else if (patternParts[i] !== hashParts[i]) {
        return null;
      }
    }
    return { params };
  }

  animateTransition(callback) {
    const container = document.getElementById('page-container');
    container.style.opacity = '0';
    container.style.transform = 'translateY(10px)';

    setTimeout(() => {
      callback();
      container.style.opacity = '1';
      container.style.transform = 'translateY(0)';
    }, 150);
  }
}

export default new Router();
```

### 5.2 localStorage 缓存策略

```javascript
// storage.js
const StorageKeys = {
  USER_PROFILE: 'lts_user_profile',
  STUDY_RECORDS: 'lts_study_records',
  POMODORO_SESSIONS: 'lts_pomodoro_sessions',
  READING_RECORDS: 'lts_reading_records',
  KNOWLEDGE_STATE: 'lts_knowledge_state',
  ACHIEVEMENTS: 'lts_achievements',
  SETTINGS: 'lts_settings',
  SYNC_META: 'lts_sync_meta',
  SEARCH_HISTORY: 'lts_search_history',
  CHART_PREFS: 'lts_chart_preferences',
  EXAMS: 'lts_exams',
};

class Storage {
  constructor() {
    this.cache = new Map(); // 内存缓存
  }

  get(key) {
    // 1. 先查内存缓存
    if (this.cache.has(key)) {
      return this.cache.get(key);
    }

    // 2. 查 localStorage
    try {
      const raw = localStorage.getItem(key);
      const data = raw ? JSON.parse(raw) : null;
      this.cache.set(key, data);
      return data;
    } catch (e) {
      console.error(`Storage read error: ${key}`, e);
      return null;
    }
  }

  set(key, value) {
    try {
      this.cache.set(key, value);
      localStorage.setItem(key, JSON.stringify(value));
      // 触发同步队列（使用防抖避免 data:changed 重复触发）
      this._emitChange(key, value);
    } catch (e) {
      console.error(`Storage write error: ${key}`, e);
      // localStorage 满时清理旧数据（使用直接操作避免递归）
      if (!this._isCleaningUp) {
        this._isCleaningUp = true;
        try { this.cleanup(); } finally { this._isCleaningUp = false; }
      }
    }
  }

  _emitChange(key, value) {
    clearTimeout(this._emitTimer);
    this._emitTimer = setTimeout(() => {
      EventBus.emit('data:changed', { key, value });
    }, 50); // 50ms 防抖，同一批次多次 set 只触发一次事件
  }

  cleanup() {
    // 直接操作 localStorage，不经过 set()，避免递归
    const records = this.get(StorageKeys.STUDY_RECORDS) || [];
    if (records.length > 10000) {
      const trimmed = records.slice(-10000);
      this.cache.set(StorageKeys.STUDY_RECORDS, trimmed);
      try {
        localStorage.setItem(StorageKeys.STUDY_RECORDS, JSON.stringify(trimmed));
      } catch (e) { /* 仍然满则放弃 */ }
    }
  }
}

export default new Storage();
```

### 5.3 GitHub API 同步

```javascript
// sync-engine.js
class SyncEngine {
  constructor() {
    this.token = null;
    this.owner = null;
    this.repo = null;
    this.syncQueue = [];
    this.isSyncing = false;
    this.rateLimitRemaining = 60; // GitHub API 速率限制
    this.rateLimitReset = 0;
  }

  configure(token, owner, repo) {
    this.token = token;
    this.owner = owner;
    this.repo = repo;
  }

  // 上传数据到 GitHub
  async upload(key, data) {
    const path = `data/${key}.json`;
    const content = btoa(unescape(encodeURIComponent(JSON.stringify(data, null, 2))));

    // 获取当前文件 SHA（如果存在）
    const sha = await this.getFileSha(path);

    const response = await fetch(
      `https://api.github.com/repos/${this.owner}/${this.repo}/contents/${path}`,
      {
        method: 'PUT',
        headers: {
          'Authorization': `token ${this.token}`,
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({
          message: `Update ${key} via LTS`,
          content,
          sha,
        }),
      }
    );

    // 速率限制检查
    this._updateRateLimit(response);

    if (!response.ok) {
      throw new Error(`GitHub API error: ${response.status}`);
    }

    return response.json();
  }

  // 从 GitHub 下载数据
  async download(key) {
    const path = `data/${key}.json`;

    const response = await fetch(
      `https://api.github.com/repos/${this.owner}/${this.repo}/contents/${path}`,
      {
        headers: {
          'Authorization': `token ${this.token}`,
        },
      }
    );

    if (response.status === 404) return null;
    if (!response.ok) throw new Error(`GitHub API error: ${response.status}`);

    const data = await response.json();
    const content = decodeURIComponent(escape(atob(data.content)));
    return JSON.parse(content);
  }

  async getFileSha(path) {
    try {
      const response = await fetch(
        `https://api.github.com/repos/${this.owner}/${this.repo}/contents/${path}`,
        { headers: { 'Authorization': `token ${this.token}` } }
      );
      this._updateRateLimit(response);
      if (response.ok) {
        const data = await response.json();
        return data.sha;
      }
    } catch (e) { /* 文件不存在 */ }
    return undefined;
  }

  _updateRateLimit(response) {
    const remaining = response.headers.get('X-RateLimit-Remaining');
    const reset = response.headers.get('X-RateLimit-Reset');
    if (remaining !== null) this.rateLimitRemaining = parseInt(remaining, 10);
    if (reset !== null) this.rateLimitReset = parseInt(reset, 10) * 1000;
  }

  async _waitIfRateLimited() {
    if (this.rateLimitRemaining <= 1) {
      const waitMs = Math.max(0, this.rateLimitReset - Date.now()) + 1000;
      if (waitMs > 0) {
        console.warn(`GitHub API rate limited, waiting ${waitMs}ms`);
        await new Promise(r => setTimeout(r, Math.min(waitMs, 60000)));
      }
    }
  }

  // 全量同步
  async fullSync() {
    if (this.isSyncing) return;
    this.isSyncing = true;

    try {
      const keys = Object.values(StorageKeys);
      for (const key of keys) {
        await this._waitIfRateLimited();
        const local = Storage.get(key);
        const remote = await this.download(key);

        if (!remote) {
          // 远端没有，上传本地数据
          if (local) await this.upload(key, local);
        } else if (!local) {
          // 本地没有，使用远端数据
          Storage.set(key, remote);
        } else {
          // 都有，合并（以最新的为准）
          const merged = this.mergeData(key, local, remote);
          Storage.set(key, merged);
          await this.upload(key, merged);
        }
      }

      Storage.set(StorageKeys.SYNC_META, {
        lastSync: Date.now(),
        status: 'success',
      });
      EventBus.emit('sync:completed', { status: 'success' });
    } catch (e) {
      Storage.set(StorageKeys.SYNC_META, {
        lastSync: Date.now(),
        status: 'error',
        error: e.message,
      });
      EventBus.emit('sync:errored', { error: e });
    } finally {
      this.isSyncing = false;
    }
  }

  mergeData(key, local, remote) {
    // 对于记录类数据，合并去重（以 ID 为键）
    if (Array.isArray(local) && Array.isArray(remote)) {
      const map = new Map();
      remote.forEach(item => map.set(item.id, item));
      local.forEach(item => map.set(item.id, item));
      return Array.from(map.values()).sort((a, b) => b.timestamp - a.timestamp);
    }
    // 对于配置类数据，以时间戳较新的为准
    return (local.updatedAt > remote.updatedAt) ? local : remote;
  }
}

export default new SyncEngine();
```

## 7. 核心算法规格

### 7.1 温度模型

**公式：**
```
T(t) = T_peak × 2^(-t / t_half)
```

| 变量 | 说明 |
|------|------|
| `T(t)` | 当前温度 |
| `T_peak` | 学习后达到的峰值温度 |
| `t` | 距上次学习的时间（天） |
| `t_half` | 半衰期（天），动态计算 |

**温度级别：**

| 级别 | 温度范围 | 颜色 | emoji |
|------|----------|------|-------|
| 炙热 | ≥ 90 | #FF4444 | 🔥 |
| 温热 | 70 - 89 | #FF8C00 | ☀️ |
| 温暖 | 50 - 69 | #FFD700 | 🌤 |
| 正常 | 30 - 49 | #87CEEB | 🌥 |
| 微凉 | 10 - 29 | #B0C4DE | ❄️ |
| 冻结 | < 10 | #D3D3D3 | 🧊 |

**动态半衰期：**
```
t_half = t_base × f_subject × f_accuracy × f_streak

约束: t_half ∈ [0.5, 30]
```

| 参数 | 说明 | 范围 |
|------|------|------|
| `t_base` | 基础半衰期（默认 7 天） | - |
| `f_subject` | 学科系数（EMA 自动拟合，见下文） | 0.8 - 1.2 |
| `f_accuracy` | 得分率系数（高分延长，低分缩短） | 0.5 - 1.5 |
| `f_streak` | 连续天数系数（连续越长越稳定） | 0.8 - 1.2 |

**学科系数 EMA 自动拟合：**

不再使用固定值（理科 1.05 / 文科 0.95），改为根据用户实际学习数据自动拟合每个学科的系数。

```javascript
// 初始值：所有学科默认 1.0
// 每次学习后，使用 EMA 根据半衰期误差自动更新
function updateSubjectCoeff(profile, subject, actualHalfLife, predictedHalfLife) {
  const alpha = 0.1; // 平滑系数
  // 计算误差比：实际半衰期 / 预测半衰期
  const errorRatio = Math.max(0.5, Math.min(2.0, actualHalfLife / predictedHalfLife));
  // EMA 更新
  const oldCoeff = profile.subjectCoeff[subject] || 1.0;
  profile.subjectCoeff[subject] = oldCoeff * (1 - alpha) + errorRatio * alpha;
  // 约束到 [0.8, 1.2]
  profile.subjectCoeff[subject] = Math.max(0.8, Math.min(1.2, profile.subjectCoeff[subject]));
}
```

| 步骤 | 说明 |
|------|------|
| 初始值 | 所有学科 `f_subject = 1.0` |
| 数据来源 | 每次知识点复习时，对比预测温度衰减与实际得分 |
| 更新频率 | 每次有效学习交互后 |
| 约束范围 | `[0.8, 1.2]`，防止极端值 |

**峰值温度计算：**
```
T_peak = min(100, T_current + ΔT)

ΔT = baseGain × qualityMultiplier × momentum
```

| 参数 | 说明 |
|------|------|
| `baseGain` | 基础增益，= min(30, durationMinutes × 0.5) |
| `qualityMultiplier` | 质量乘数，基于得分率 (0.5 - 1.5) |
| `momentum` | 动量系数 (-0.3 到 0.3) |

**温度定义总结：** 温度是衰减指标，范围 0-100，反映"当前记忆的新鲜程度"。每次学习后温度上升（T_peak），之后随时间指数衰减：`T(t) = T_peak × 2^(-t / t_half)`。

### 7.1.1 掌握度模型

**定义：** 掌握度是累积指标，范围 0-1，反映"对知识点的整体掌握水平"。与温度不同，掌握度不会随时间自然衰减，只在新的学习交互中更新。

**公式：**
```
mastery_new = mastery_old × (1 - α) + scoreRate × α

α = 0.15 × weight
weight = min(2.0, 1 + studyCount × 0.1)
```

| 变量 | 说明 |
|------|------|
| `mastery_new` | 更新后的掌握度 |
| `mastery_old` | 更新前的掌握度 |
| `scoreRate` | 本次学习得分率 (0-1) |
| `α` | EMA 平滑系数，基于学习次数动态调整 |
| `studyCount` | 累计学习次数 |

**温度 vs 掌握度：**

| 维度 | 温度 (Temperature) | 掌握度 (Mastery) |
|------|--------------------|--------------------|
| 范围 | 0 - 100 | 0 - 1 |
| 类型 | 衰减指标 | 累积指标 |
| 时间衰减 | 是（指数衰减） | 否 |
| 用途 | 决定复习优先级（越冷越需要复习） | 衡量长期能力水平 |
| 变化速度 | 快（几天不学就下降） | 慢（稳步缓慢积累） |
| 决策权重 | 阴影队列排序、背包算法 | 等级评估、称号判定 |

### 7.2 XP 引擎 2.0

**公式：**
```
XP = round(rawXP × quality × decay × softCap)
```

**rawXP 计算：**
```
rawXP = durationMinutes × subjectCoeff × difficultyCoeff
```

| 参数 | 说明 | 默认值 |
|------|------|--------|
| `durationMinutes` | 学习时长（分钟） | - |
| `subjectCoeff` | 学科系数 | 1.0 |
| `difficultyCoeff` | 难度系数（基于知识点温度，越冷越难） | 0.8 - 1.2 |

**quality 质量乘数：**
```
scoreRate_clamped = clamp(scoreRate, 0, 1)
quality = clamp(1 + (scoreRate_clamped - 0.5) × 0.4, 0.8, 1.2)
```
- 得分率 100%: quality = 1.2
- 得分率 50%: quality = 1.0
- 得分率 0%: quality = 0.8
- 边界保护：scoreRate 强制 clamp 到 [0, 1]，quality 输出 clamp 到 [0.8, 1.2]

**decay 衰减系数：**
```
decay = clamp(1 - todayXP / dailyCap, 0.5, 1.0)
```
- `dailyCap` = 用户等级 × 100 + 500
- 防止单日无限刷 XP
- 最终 XP 保底：`XP = max(1, round(rawXP × quality × decay × softCap))`

**softCap 软上限：**
```
if (todayXP > dailyCap * 0.8) {
  raw = 0.5 + 0.5 × (1 - (todayXP - dailyCap * 0.8) / (dailyCap * 0.2));
  softCap = Math.max(0, raw);  // 防止负值
} else {
  softCap = 1.0;
}
```

### 7.3 阴影队列（Shadow Queue）

**优先级计算：**
```
urgency = (80 - currentTemp) × examWeight / max(1, daysUntilExam)
```

| 变量 | 说明 |
|------|------|
| `currentTemp` | 当前温度（越低越紧急） |
| `examWeight` | 考试权重 (0-1) |
| `daysUntilExam` | 距离考试天数 |

**队列排序：** 按 urgency 降序排列，每小时重新计算一次。

### 7.4 背包算法（Knapsack）

用于在时间预算内选择最优的复习组合：

**0/1 背包 DP：**
```
dp[i][t] = max(dp[i-1][t], dp[i-1][t-cost_i] + benefit_i)
```

| 变量 | 说明 |
|------|------|
| `i` | 知识点索引 |
| `t` | 时间预算（分钟） |
| `cost_i` | 该知识点的复习时长（预估），至少 1 分钟 |
| `benefit_i` | 复习收益 = urgency × (1 - mastery)，至少 0 |

**边界保护：**
- 时间预算 `t` 为 0 或负数时，返回空列表
- 候选知识点列表为空时，返回空列表
- `cost_i` 为 0 时跳过该项（避免除零）
- `benefit_i` 为负数时置 0

```javascript
function knapsackRecommend(knowledgeState, timeBudget) {
  const candidates = Object.values(knowledgeState)
    .map(k => ({
      ...k,
      cost: Math.max(1, Math.ceil(k.totalMinutes / Math.max(1, k.studyCount) * 0.5)),
      benefit: Math.max(0, calculateUrgency(k) * (1 - k.mastery)),
    }))
    .filter(k => k.cost <= timeBudget && k.benefit > 0);

  if (candidates.length === 0 || timeBudget <= 0) return [];

  // 0/1 背包 DP
  const n = candidates.length;
  const W = Math.floor(timeBudget);
  const dp = Array.from({ length: n + 1 }, () => new Float64Array(W + 1));

  for (let i = 1; i <= n; i++) {
    const { cost, benefit } = candidates[i - 1];
    for (let t = 0; t <= W; t++) {
      dp[i][t] = dp[i - 1][t];
      if (cost <= t) {
        dp[i][t] = Math.max(dp[i][t], dp[i - 1][t - cost] + benefit);
      }
    }
  }

  // 回溯选中项
  const result = [];
  let t = W;
  for (let i = n; i > 0; i--) {
    if (dp[i][t] !== dp[i - 1][t]) {
      result.push(candidates[i - 1]);
      t -= candidates[i - 1].cost;
    }
  }
  return result;
}
```

**输出：** 在给定时间预算内，返回最值得复习的知识点列表。

### 7.5 动量（Momentum）

**计算方式：** 最近 10 条学习记录的线性回归斜率，tanh 压缩到 [-0.3, 0.3]。

```javascript
function calculateMomentum(records) {
  if (records.length < 3) return 0;

  // 取最近 10 条
  const recent = records.slice(-10);
  const n = recent.length;

  // 线性回归: y = mx + b
  const xMean = (n - 1) / 2;
  const yMean = recent.reduce((s, r) => s + r.scoreRate, 0) / n;

  let numerator = 0, denominator = 0;
  for (let i = 0; i < n; i++) {
    numerator += (i - xMean) * (recent[i].scoreRate - yMean);
    denominator += (i - xMean) ** 2;
  }

  const slope = denominator ? numerator / denominator : 0;

  // tanh 压缩到 [-0.3, 0.3]
  return Math.tanh(slope * 5) * 0.3;
}
```

### 7.6 假性熟练检测

**三个条件同时满足时判定为假性熟练：**

```javascript
function isFalseMastery(knowledge) {
  return (
```

```javascript
function updateProfile(profile, newRecord, knowledgeState) {
  const alpha = 0.1; // 平滑系数

  // 学科能力更新
  const subject = newRecord.subject;
  const oldAbility = profile.subjectAbility[subject] || 0.5;
  profile.subjectAbility[subject] =
    oldAbility * (1 - alpha) + newRecord.scoreRate * alpha;

  // 时段系数更新
  const hour = new Date(newRecord.timestamp).getHours();
  const period = getPeriod(hour); // morning/afternoon/evening/night
  const oldMod = profile.modifiers.timeOfDay[period];
  profile.modifiers.timeOfDay[period] =
    oldMod * (1 - alpha) + (newRecord.scoreRate > 0.7 ? 1.1 : 0.9) * alpha;

  // 全局半衰期更新
  const avgMastery = Object.values(knowledgeState)
    .reduce((s, k) => s + k.mastery, 0) / Object.keys(knowledgeState).length;
  profile.globalHalfLife = 5 + avgMastery * 10; // [5, 15]

  profile.updatedAt = Date.now();
  return profile;
}
```

```javascript
function levelToScoreRate(level) {
  // 使用 sigmoid 函数拟合
  // Level 1-10: 0-30%
  // Level 11-20: 30-50%
  // ...
  // Level 91-100: 98-100%
  // 参数: k=0.12 控制曲线陡峭度, x0=50 为中心点
  return 1 / (1 + Math.exp(-0.12 * (level - 50)));
}

function scoreRateToLevel(scoreRate) {
  // 反函数 — 边界保护
  const clamped = Math.max(0.01, Math.min(0.99, scoreRate));
  return Math.max(1, Math.min(100, Math.round(50 - Math.log(1 / clamped - 1) / 0.12)));
}
```

### 8.1 事件总线

```javascript
// event-bus.js
class EventBus {
  constructor() {
    this.listeners = new Map();
  }

  on(event, callback) {
    if (!this.listeners.has(event)) {
      this.listeners.set(event, []);
    }
    this.listeners.get(event).push(callback);
    return () => this.off(event, callback);
  }

  off(event, callback) {
    const cbs = this.listeners.get(event);
    if (cbs) {
      const idx = cbs.indexOf(callback);
      if (idx > -1) cbs.splice(idx, 1);
    }
  }

  emit(event, data) {
    const cbs = this.listeners.get(event);
    if (cbs) {
      cbs.forEach(cb => {
        try { cb(data); }
        catch (e) { console.error(`Event handler error [${event}]:`, e); }
      });
    }
  }
}

export default new EventBus();
```

```javascript
  }

  close() {
    this.element.classList.remove('bottom-sheet--open');
    setTimeout(() => {
      this.element.remove();
      this.element = null;
      this.isOpen = false;
    }, 300);
  }

  initDrag() {
    const handle = this.element.querySelector('.bottom-sheet__handle');
    const content = this.element.querySelector('.bottom-sheet__content');
    let startY, currentY;

    handle.addEventListener('touchstart', (e) => {
      startY = e.touches[0].clientY;
    });

    handle.addEventListener('touchmove', (e) => {
      currentY = e.touches[0].clientY;
      const diff = currentY - startY;
      if (diff > 0) {
        content.style.transform = `translateY(${diff}px)`;
      }
    });

    handle.addEventListener('touchend', () => {
      // 检查 startY/currentY 是否已定义（touchend 可能在无 touchmove 时触发）
      if (startY == null || currentY == null) return;
      const diff = currentY - startY;
      if (diff > 100) {
        this.close();
      } else {
        content.style.transform = '';
      }
      startY = null;
      currentY = null;
    });
  }
}
```

```javascript
class VirtualList {
  constructor(container, items, itemHeight, renderItem) {
    this.container = container;
    this.items = items;
    this.itemHeight = itemHeight;
    this.renderItem = renderItem;
    this.visibleCount = Math.ceil(container.clientHeight / itemHeight) + 2;
    this.scrollTop = 0;

    this.init();
  }

  init() {
    this.container.style.overflow = 'auto';
    this.container.style.position = 'relative';

    this.spacer = document.createElement('div');
    this.spacer.style.height = `${this.items.length * this.itemHeight}px`;
    this.container.appendChild(this.spacer);

    this.viewport = document.createElement('div');
    this.viewport.style.position = 'absolute';
    this.viewport.style.left = '0';
    this.viewport.style.right = '0';
    this.container.appendChild(this.viewport);

    this.container.addEventListener('scroll', () => {
      this.scrollTop = this.container.scrollTop;
      this.render();
    });

    this.render();
  }

  render() {
    const startIndex = Math.floor(this.scrollTop / this.itemHeight);
    const endIndex = Math.min(startIndex + this.visibleCount, this.items.length);

    this.viewport.style.top = `${startIndex * this.itemHeight}px`;
    this.viewport.innerHTML = '';

    for (let i = startIndex; i < endIndex; i++) {
      const el = this.renderItem(this.items[i], i);
      el.style.height = `${this.itemHeight}px`;
      this.viewport.appendChild(el);
    }
  }
}
```

### 11.2 计时器实现

```javascript
// timer.js
class PomodoroTimer {
  constructor() {
    this.state = 'idle'; // idle | focusing | paused | break
    this.remaining = 0;
    this.total = 0;
    this.preset = null;
    this.completedCount = 0;
    this.intervalId = null;
    this.startTime = null;
    this.pausedTime = 0;
  }

  start(preset, subject) {
    this.preset = preset;
    this.subject = subject;
    this.total = preset.work * 60;  // preset.work 为专注时长（分钟）
    this.remaining = this.total;
    this.state = 'focusing';
    this.startTime = Date.now();

    this.tick();
    this.intervalId = setInterval(() => this.tick(), 1000);

    EventBus.emit('pomodoro:started', { preset, subject });
  }

  tick() {
    if (this.state === 'focusing' || this.state === 'break') {
      const now = Date.now();
      const elapsed = Math.floor((now - this.startTime - this.pausedTime) / 1000);
      this.remaining = Math.max(0, this.total - elapsed);

      EventBus.emit('pomodoro:tick', {
        remaining: this.remaining,
        total: this.total,
        state: this.state,
      });

      if (this.remaining <= 0) {
        this.complete();
      }
    }
  }

  pause() {
    if (this.state !== 'focusing') return;
    this.state = 'paused';
    this.pauseStart = Date.now();
    clearInterval(this.intervalId);
    EventBus.emit('pomodoro:paused', {});
  }

  resume() {
    if (this.state !== 'paused') return;
    this.state = 'focusing';
    this.pausedTime += Date.now() - this.pauseStart;
    this.intervalId = setInterval(() => this.tick(), 1000);
    EventBus.emit('pomodoro:resumed', {});
  }

  complete() {
    clearInterval(this.intervalId);

    if (this.state === 'focusing') {
      this.completedCount++;
      const isLongBreak = this.completedCount % 4 === 0;
      const breakDuration = isLongBreak
        ? this.preset.longBreak
        : this.preset.break;

      // 生成学习记录
      this.createRecord();

      EventBus.emit('pomodoro:completed', {
        completedCount: this.completedCount,
        workMinutes: this.preset.work,
      });

      // 自动开始休息
      this.startBreak(breakDuration, isLongBreak);
    } else if (this.state === 'break') {
      EventBus.emit('pomodoro:break:ended', {});
      this.state = 'idle';
    }
  }

  startBreak(minutes, isLong) {
    this.state = 'break';
    this.total = minutes * 60;
    this.remaining = this.total;
    this.startTime = Date.now();
    this.pausedTime = 0;

    this.tick();
    this.intervalId = setInterval(() => this.tick(), 1000);

    EventBus.emit('pomodoro:break:started', { isLong, duration: minutes });
  }

  createRecord() {
    const record = {
      id: `rec_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`,
      timestamp: Date.now(),
      date: formatDate(new Date()),
      subject: this.subject,
      durationMinutes: this.preset.work,
      source: 'pomodoro',
      pomodoroSessionId: this.sessionId,
    };

    EventBus.emit('record:created', { record });
    return record;
  }

  // 手动停止番茄钟
  // 90% 阈值规则：若已完成 ≥90% 的专注时长，仍计为一次有效番茄并生成记录
  stop() {
    clearInterval(this.intervalId);
    const elapsed = this.total - this.remaining;
    const threshold = this.total * 0.9;

    if (this.state === 'focusing' && elapsed >= threshold) {
      this.completedCount++;
      this.createRecord();
    }

    this.state = 'idle';
    this.remaining = 0;
    EventBus.emit('pomodoro:stopped', { elapsed, threshold: elapsed >= threshold });
  }

  formatTime(seconds) {
    const m = Math.floor(seconds / 60);
    const s = seconds % 60;
    return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
  }
}

export default new PomodoroTimer();
```

```javascript
// search-index.js
class SearchIndex {
  constructor() {
    this.index = new Map(); // keyword → [{ id, type, field, weight }]
  }

  // 构建索引
  build(data) {
    this.index.clear();

    // 学科
    data.subjects.forEach(s => {
      this.addEntry(s.name, s.id, 'subject', 'name', 10);
      this.addEntry(toPinyin(s.name), s.id, 'subject', 'pinyin', 5);
      this.addEntry(toPinyinInitials(s.name), s.id, 'subject', 'initials', 8);
    });

    // 知识点
    Object.values(data.knowledgeState).forEach(k => {
      this.addEntry(k.name, k.id, 'knowledge', 'name', 8);
      this.addEntry(toPinyin(k.name), k.id, 'knowledge', 'pinyin', 4);
    });

    // 学习记录备注
    data.records.forEach(r => {
      if (r.notes) {
        this.addEntry(r.notes, r.id, 'record', 'notes', 3);
      }
    });

    // 功能入口
    const functions = [
      { name: '设置', route: '#/settings', icon: '⚙️' },
      { name: '数据图表', route: '#/data/charts', icon: '📈' },
      { name: '导出数据', route: '#/settings', icon: '📤' },
      // ...
    ];
    functions.forEach(f => {
      this.addEntry(f.name, f.route, 'function', 'name', 6);
    });
  }

  addEntry(text, id, type, field, weight) {
    const tokens = this.tokenize(text);
    tokens.forEach(token => {
      if (!this.index.has(token)) {
        this.index.set(token, []);
      }
      this.index.get(token).push({ id, type, field, weight });
    });
  }

  tokenize(text) {
    // 分词：按空格、标点分割 + n-gram
    const tokens = new Set();
    const clean = text.toLowerCase().replace(/[^\w一-鿿]/g, '');

    // 完整词
    tokens.add(clean);

    // 2-gram
    for (let i = 0; i < clean.length - 1; i++) {
      tokens.add(clean.slice(i, i + 2));
    }

    // 3-gram
    for (let i = 0; i < clean.length - 2; i++) {
      tokens.add(clean.slice(i, i + 3));
    }

    return tokens;
  }

  // 搜索
  search(query, limit = 20) {
    const tokens = this.tokenize(query.toLowerCase());
    const scores = new Map(); // id → { score, type, id }

    tokens.forEach(token => {
      // 精确匹配
      if (this.index.has(token)) {
        this.index.get(token).forEach(entry => {
          const key = `${entry.type}:${entry.id}`;
          const existing = scores.get(key) || { score: 0, ...entry };
          existing.score += entry.weight;
          scores.set(key, existing);
        });
      }

      // 前缀匹配
      for (const [key, entries] of this.index) {
        if (key.startsWith(token) && key !== token) {
          entries.forEach(entry => {
            const k = `${entry.type}:${entry.id}`;
            const existing = scores.get(k) || { score: 0, ...entry };
            existing.score += entry.weight * 0.5; // 前缀匹配权重减半
            scores.set(k, existing);
          });
        }
      }
    });

    return Array.from(scores.values())
      .sort((a, b) => b.score - a.score)
      .slice(0, limit);
  }
}

export default new SearchIndex();
```

```javascript
  };
}

export function throttle(fn, limit = 100) {
  let inThrottle;
  return function (...args) {
    if (!inThrottle) {
      fn.apply(this, args);
      inThrottle = true;
      setTimeout(() => (inThrottle = false), limit);

  };
```

```javascript
    this.cache = new Map();
    this.defaultTTL = defaultTTL;
  }

  set(key, value, ttl = this.defaultTTL) {
```

```javascript
      request.headers.get('Authorization')) {

        caches.open(CACHE_NAME).then((cache) => cache.put(request, clone));
        return response;
      });
    })
```

### 15.2 网络重试策略

```javascript
async function fetchWithRetry(url, options = {}, maxRetries = 3) {
  for (let i = 0; i < maxRetries; i++) {
    try {
      const response = await fetch(url, options);
      if (response.ok) return response;

      // 429 限流：指数退避
      if (response.status === 429) {
        const delay = Math.pow(2, i) * 1000 + Math.random() * 1000;
        await new Promise((r) => setTimeout(r, delay));
        continue;
      }

      // 5xx 服务器错误：可重试
      if (response.status >= 500 && i < maxRetries - 1) {
        const delay = Math.pow(2, i) * 1000;
        await new Promise((r) => setTimeout(r, delay));
        continue;
      }

      // 4xx 客户端错误：不重试
      throw new Error(`HTTP ${response.status}`);
    } catch (error) {
      if (i === maxRetries - 1) throw error;
      const delay = Math.pow(2, i) * 1000;
      await new Promise((r) => setTimeout(r, delay));
    }
  }
}
```

### B. 命名约定

| 类型 | 规范 | 示例 |
|------|------|------|
| 文件名 | kebab-case | `data-engine.js` |
| CSS 类名 | BEM 变体（实际使用 `block__element--modifier`） | `.card__title--large`, `.bottom-sheet__handle` |
| CSS 类名前缀 | 组件名作为 block，子元素用 `__`，状态用 `--` | `.toast--show`, `.progress-bar__fill` |
| JS 变量 | camelCase | `studyRecords` |
| JS 常量 | UPPER_SNAKE | `MAX_RETRY_COUNT` |
| JS 类名 | PascalCase | `PomodoroTimer` |
| 事件名 | `名词:过去分词`（持续性/状态性事件除外） | `pomodoro:completed`, `data:changed`, `exam:created` |
| CSS 变量 | kebab-case, 语义化全称 | `--color-text-primary`, `--space-md`, `--radius-sm` |
