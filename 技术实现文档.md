## 4. 路由系统

### 4.1 Hash 路由设计

使用 `hashchange` 事件实现路由，不依赖 History API（兼容性更好）：

```javascript
// 路由表
const routes = {
  '#/':           'home',
  '#/data':       'data-tab',
  '#/data/charts': 'data-charts',
  '#/data/log':   'data-log',

  '#/data/reading': 'data-reading',
  '#/data/skill-tree': 'skill-tree',
  '#/pomodoro':   'pomodoro-fullscreen',
  '#/settings':   'settings',
  '#/about':      'about',
  '#/achievement': 'achievement',
  '#/subject/:id': 'subject-detail',
  '#/search':     'search',
};
```

### 4.2 Router 实现

```javascript
// router.js
class Router {
  constructor() {
    this.routes = new Map();
    this.currentRoute = null;
    this.history = [];

    window.addEventListener('hashchange', () => this.handleRoute());
    window.addEventListener('load', () => this.handleRoute());
  }

  register(pattern, handler) {
    this.routes.set(pattern, handler);
  }

  navigate(hash) {
    window.location.hash = hash;
  }

  back() {
    if (this.history.length > 1) {
      this.history.pop();
      window.location.hash = this.history[this.history.length - 1];
    } else {
      window.location.hash = '#/';
    }
  }

  handleRoute() {
    const hash = window.location.hash || '#/';
    const [path, ...params] = hash.split('/').filter(Boolean);

    // 匹配路由
    for (const [pattern, handler] of this.routes) {
      const match = this.matchPattern(pattern, hash);
      if (match) {
        // 页面切换动画
        this.animateTransition(() => {
          handler(match.params);
          this.currentRoute = hash;
          this.history.push(hash);
        });
        return;
      }
    }

    // 404 回首页
    this.navigate('#/');
  }

  matchPattern(pattern, hash) {
    // 支持 :id 参数匹配
    const patternParts = pattern.split('/');
    const hashParts = hash.split('/');

    if (patternParts.length !== hashParts.length) return null;

    const params = {};
    for (let i = 0; i < patternParts.length; i++) {
      if (patternParts[i].startsWith(':')) {
        params[patternParts[i].slice(1)] = hashParts[i];
      } else if (patternParts[i] !== hashParts[i]) {
        return null;
      }
    }
    return { params };
  }

  animateTransition(callback) {
    const container = document.getElementById('page-container');
    container.style.opacity = '0';
    container.style.transform = 'translateY(10px)';

    setTimeout(() => {
      callback();
      container.style.opacity = '1';
      container.style.transform = 'translateY(0)';
    }, 150);
  }
}

export default new Router();
```

### 5.1 双层存储架构


**策略：** localStorage 是缓存/保底，GitHub 是主存储。

### 5.2 localStorage 缓存策略

```javascript
// storage.js
const StorageKeys = {
  USER_PROFILE: 'lts_user_profile',
  STUDY_RECORDS: 'lts_study_records',
  POMODORO_SESSIONS: 'lts_pomodoro_sessions',
  READING_RECORDS: 'lts_reading_records',
  KNOWLEDGE_STATE: 'lts_knowledge_state',
  ACHIEVEMENTS: 'lts_achievements',
  SETTINGS: 'lts_settings',
  SYNC_META: 'lts_sync_meta',
  SEARCH_HISTORY: 'lts_search_history',
  CHART_PREFS: 'lts_chart_preferences',
};

class Storage {
  constructor() {
    this.cache = new Map(); // 内存缓存
  }

  get(key) {
    // 1. 先查内存缓存
    if (this.cache.has(key)) {
      return this.cache.get(key);
    }

    // 2. 查 localStorage
    try {
      const raw = localStorage.getItem(key);
      const data = raw ? JSON.parse(raw) : null;
      this.cache.set(key, data);
      return data;
    } catch (e) {
      console.error(`Storage read error: ${key}`, e);
      return null;
    }
  }

  set(key, value) {
    try {
      this.cache.set(key, value);
      localStorage.setItem(key, JSON.stringify(value));
      // 触发同步队列
      EventBus.emit('data:changed', { key, value });
    } catch (e) {
      console.error(`Storage write error: ${key}`, e);
      // localStorage 满时清理旧数据
      this.cleanup();
    }
  }

  cleanup() {
    // 清理最旧的学习记录（保留最近 10000 条）
    const records = this.get(StorageKeys.STUDY_RECORDS) || [];
    if (records.length > 10000) {
      this.set(StorageKeys.STUDY_RECORDS, records.slice(-10000));
    }
  }
}

export default new Storage();
```

### 5.3 GitHub API 同步

```javascript
// sync-engine.js
class SyncEngine {
  constructor() {
    this.token = null;
    this.owner = null;
    this.repo = null;
    this.syncQueue = [];
    this.isSyncing = false;
  }

  configure(token, owner, repo) {
    this.token = token;
    this.owner = owner;
    this.repo = repo;
  }

  // 上传数据到 GitHub
  async upload(key, data) {
    const path = `data/${key}.json`;
    const content = btoa(unescape(encodeURIComponent(JSON.stringify(data, null, 2))));

    // 获取当前文件 SHA（如果存在）
    const sha = await this.getFileSha(path);

    const response = await fetch(
      `https://api.github.com/repos/${this.owner}/${this.repo}/contents/${path}`,
      {
        method: 'PUT',
        headers: {
          'Authorization': `token ${this.token}`,
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({
          message: `Update ${key} via LTS`,
          content,
          sha,
        }),
      }
    );

    if (!response.ok) {
      throw new Error(`GitHub API error: ${response.status}`);
    }

    return response.json();
  }

  // 从 GitHub 下载数据
  async download(key) {
    const path = `data/${key}.json`;

    const response = await fetch(
      `https://api.github.com/repos/${this.owner}/${this.repo}/contents/${path}`,
      {
        headers: {
          'Authorization': `token ${this.token}`,
        },
      }
    );

    if (response.status === 404) return null;
    if (!response.ok) throw new Error(`GitHub API error: ${response.status}`);

    const data = await response.json();
    const content = decodeURIComponent(escape(atob(data.content)));
    return JSON.parse(content);
  }

  async getFileSha(path) {
    try {
      const response = await fetch(
        `https://api.github.com/repos/${this.owner}/${this.repo}/contents/${path}`,
        { headers: { 'Authorization': `token ${this.token}` } }
      );
      if (response.ok) {
        const data = await response.json();
        return data.sha;
      }
    } catch (e) { /* 文件不存在 */ }
    return undefined;
  }

  // 全量同步
  async fullSync() {
    if (this.isSyncing) return;
    this.isSyncing = true;

    try {
      const keys = Object.values(StorageKeys);
      for (const key of keys) {
        const local = Storage.get(key);
        const remote = await this.download(key);

        if (!remote) {
          // 远端没有，上传本地数据
          if (local) await this.upload(key, local);
        } else if (!local) {
          // 本地没有，使用远端数据
          Storage.set(key, remote);
        } else {
          // 都有，合并（以最新的为准）
          const merged = this.mergeData(key, local, remote);
          Storage.set(key, merged);
          await this.upload(key, merged);
        }
      }

      Storage.set(StorageKeys.SYNC_META, {
        lastSync: Date.now(),
        status: 'success',
      });
      EventBus.emit('sync:complete', { status: 'success' });
    } catch (e) {
      Storage.set(StorageKeys.SYNC_META, {
        lastSync: Date.now(),
        status: 'error',
        error: e.message,
      });
      EventBus.emit('sync:error', { error: e });
    } finally {
      this.isSyncing = false;
    }
  }

  mergeData(key, local, remote) {
    // 对于记录类数据，合并去重（以 ID 为键）
    if (Array.isArray(local) && Array.isArray(remote)) {
      const map = new Map();
      remote.forEach(item => map.set(item.id, item));
      local.forEach(item => map.set(item.id, item));
      return Array.from(map.values()).sort((a, b) => b.timestamp - a.timestamp);
    }
    // 对于配置类数据，以时间戳较新的为准
    return (local.updatedAt > remote.updatedAt) ? local : remote;
  }
}

export default new SyncEngine();
```

### 5.4 数据加载策略

```javascript
    this.loading = new Map();
  }

  async getData(key) {
    // 1. 立即返回 localStorage 缓存（同步）
    const cached = Storage.get(key);

    if (SyncEngine.token && !this.loading.has(key)) {
```

## 7. 核心算法规格

### 7.1 温度模型

**公式：**
```
T(t) = T_peak × 2^(-t / t_half)
```

| 变量 | 说明 |
|------|------|
| `T(t)` | 当前温度 |
| `T_peak` | 学习后达到的峰值温度 |
| `t` | 距上次学习的时间（天） |
| `t_half` | 半衰期（天），动态计算 |

**温度级别：**

| 级别 | 温度范围 | 颜色 | emoji |
|------|----------|------|-------|
| 炙热 | ≥ 90 | #FF4444 | 🔥 |
| 温热 | 70 - 89 | #FF8C00 | ☀️ |
| 温暖 | 50 - 69 | #FFD700 | 🌤 |
| 正常 | 30 - 49 | #87CEEB | 🌥 |
| 微凉 | 10 - 29 | #B0C4DE | ❄️ |
| 冻结 | < 10 | #D3D3D3 | 🧊 |

**动态半衰期：**
```
t_half = t_base × f_subject × f_accuracy × f_streak

约束: t_half ∈ [0.5, 30]
```

| 参数 | 说明 | 范围 |
|------|------|------|
| `t_base` | 基础半衰期（默认 7 天） | - |
| `f_subject` | 学科系数（理科 1.05，文科 0.95） | 0.8 - 1.2 |
| `f_accuracy` | 得分率系数（高分延长，低分缩短） | 0.5 - 1.5 |
| `f_streak` | 连续天数系数（连续越长越稳定） | 0.8 - 1.2 |

**峰值温度计算：**
```
T_peak = min(100, T_current + ΔT)

ΔT = baseGain × qualityMultiplier × momentum
```

| 参数 | 说明 |
|------|------|
| `baseGain` | 基础增益，= min(30, durationMinutes × 0.5) |
| `qualityMultiplier` | 质量乘数，基于得分率 (0.5 - 1.5) |
| `momentum` | 动量系数 (-0.3 到 0.3) |

### 7.2 XP 引擎 2.0

**公式：**
```
XP = round(rawXP × quality × decay × softCap)
```

**rawXP 计算：**
```
rawXP = durationMinutes × subjectCoeff × difficultyCoeff
```

| 参数 | 说明 | 默认值 |
|------|------|--------|
| `durationMinutes` | 学习时长（分钟） | - |
| `subjectCoeff` | 学科系数 | 1.0 |
| `difficultyCoeff` | 难度系数（基于知识点温度，越冷越难） | 0.8 - 1.2 |

**quality 质量乘数：**
```
quality = 1 + (scoreRate - 0.5) × 0.4
```
- 得分率 100%: quality = 1.2
- 得分率 50%: quality = 1.0
- 得分率 0%: quality = 0.8

**decay 衰减系数：**
```
decay = max(0.5, 1 - todayXP / dailyCap)
```
- `dailyCap` = 用户等级 × 100 + 500
- 防止单日无限刷 XP

**softCap 软上限：**
```
if (todayXP > dailyCap * 0.8) {
  softCap = 0.5 + 0.5 × (1 - (todayXP - dailyCap * 0.8) / (dailyCap * 0.2));
} else {
  softCap = 1.0;
}
```

### 7.3 阴影队列（Shadow Queue）

**优先级计算：**
```
urgency = (80 - currentTemp) × examWeight / max(1, daysUntilExam)
```

| 变量 | 说明 |
|------|------|
| `currentTemp` | 当前温度（越低越紧急） |
| `examWeight` | 考试权重 (0-1) |
| `daysUntilExam` | 距离考试天数 |

**队列排序：** 按 urgency 降序排列，每小时重新计算一次。

### 7.4 背包算法（Knapsack）

用于在时间预算内选择最优的复习组合：

**0/1 背包 DP：**
```
dp[i][t] = max(dp[i-1][t], dp[i-1][t-cost_i] + benefit_i)
```

| 变量 | 说明 |
|------|------|
| `i` | 知识点索引 |
| `t` | 时间预算（分钟） |
| `cost_i` | 该知识点的复习时长（预估） |
| `benefit_i` | 复习收益 = urgency × (1 - mastery) |

**输出：** 在给定时间预算内，返回最值得复习的知识点列表。

### 7.5 动量（Momentum）

**计算方式：** 最近 10 条学习记录的线性回归斜率，tanh 压缩到 [-0.3, 0.3]。

```javascript
function calculateMomentum(records) {
  if (records.length < 3) return 0;

  // 取最近 10 条
  const recent = records.slice(-10);
  const n = recent.length;

  // 线性回归: y = mx + b
  const xMean = (n - 1) / 2;
  const yMean = recent.reduce((s, r) => s + r.scoreRate, 0) / n;

  let numerator = 0, denominator = 0;
  for (let i = 0; i < n; i++) {
    numerator += (i - xMean) * (recent[i].scoreRate - yMean);
    denominator += (i - xMean) ** 2;
  }

  const slope = denominator ? numerator / denominator : 0;

  // tanh 压缩到 [-0.3, 0.3]
  return Math.tanh(slope * 5) * 0.3;
}
```

### 8.1 事件总线

```javascript
// event-bus.js
class EventBus {
  constructor() {
    this.listeners = new Map();
  }

  on(event, callback) {
    if (!this.listeners.has(event)) {
      this.listeners.set(event, []);
    }
    this.listeners.get(event).push(callback);
    return () => this.off(event, callback);
  }

  off(event, callback) {
    const cbs = this.listeners.get(event);
    if (cbs) {
      const idx = cbs.indexOf(callback);
      if (idx > -1) cbs.splice(idx, 1);
    }
  }

  emit(event, data) {
    const cbs = this.listeners.get(event);
    if (cbs) {
      cbs.forEach(cb => {
        try { cb(data); }
        catch (e) { console.error(`Event handler error [${event}]:`, e); }
      });
    }
  }
}

export default new EventBus();
```

```javascript
class VirtualList {
  constructor(container, items, itemHeight, renderItem) {
    this.container = container;
    this.items = items;
    this.itemHeight = itemHeight;
    this.renderItem = renderItem;
    this.visibleCount = Math.ceil(container.clientHeight / itemHeight) + 2;
    this.scrollTop = 0;

    this.init();
  }

  init() {
    this.container.style.overflow = 'auto';
    this.container.style.position = 'relative';

    this.spacer = document.createElement('div');
    this.spacer.style.height = `${this.items.length * this.itemHeight}px`;
    this.container.appendChild(this.spacer);

    this.viewport = document.createElement('div');
    this.viewport.style.position = 'absolute';
    this.viewport.style.left = '0';
    this.viewport.style.right = '0';
    this.container.appendChild(this.viewport);

    this.container.addEventListener('scroll', () => {
      this.scrollTop = this.container.scrollTop;
      this.render();
    });

    this.render();
  }

  render() {
    const startIndex = Math.floor(this.scrollTop / this.itemHeight);
    const endIndex = Math.min(startIndex + this.visibleCount, this.items.length);

    this.viewport.style.top = `${startIndex * this.itemHeight}px`;
    this.viewport.innerHTML = '';

    for (let i = startIndex; i < endIndex; i++) {
      const el = this.renderItem(this.items[i], i);
      el.style.height = `${this.itemHeight}px`;
      this.viewport.appendChild(el);
    }
  }
}
```

### 11.2 计时器实现

```javascript
// timer.js
class PomodoroTimer {
  constructor() {
    this.state = 'idle'; // idle | focusing | paused | break
    this.remaining = 0;
    this.total = 0;
    this.preset = null;
    this.completedCount = 0;
    this.intervalId = null;
    this.startTime = null;
    this.pausedTime = 0;
  }

  start(preset, subject) {
    this.preset = preset;
    this.subject = subject;
    this.total = preset.focus * 60;
    this.remaining = this.total;
    this.state = 'focusing';
    this.startTime = Date.now();

    this.tick();
    this.intervalId = setInterval(() => this.tick(), 1000);

    EventBus.emit('pomodoro:start', { preset, subject });
  }

  tick() {
    if (this.state === 'focusing' || this.state === 'break') {
      const now = Date.now();
      const elapsed = Math.floor((now - this.startTime - this.pausedTime) / 1000);
      this.remaining = Math.max(0, this.total - elapsed);

      EventBus.emit('pomodoro:tick', {
        remaining: this.remaining,
        total: this.total,
        state: this.state,
      });

      if (this.remaining <= 0) {
        this.complete();
      }
    }
  }

  pause() {
    if (this.state !== 'focusing') return;
    this.state = 'paused';
    this.pauseStart = Date.now();
    clearInterval(this.intervalId);
    EventBus.emit('pomodoro:pause', {});
  }

  resume() {
    if (this.state !== 'paused') return;
    this.state = 'focusing';
    this.pausedTime += Date.now() - this.pauseStart;
    this.intervalId = setInterval(() => this.tick(), 1000);
    EventBus.emit('pomodoro:resume', {});
  }

  complete() {
    clearInterval(this.intervalId);

    if (this.state === 'focusing') {
      this.completedCount++;
      const isLongBreak = this.completedCount % 4 === 0;
      const breakDuration = isLongBreak
        ? this.preset.longBreak
        : this.preset.break;

      // 生成学习记录
      this.createRecord();

      EventBus.emit('pomodoro:complete', {
        completedCount: this.completedCount,
        focusMinutes: this.preset.focus,
      });

      // 自动开始休息
      this.startBreak(breakDuration, isLongBreak);
    } else if (this.state === 'break') {
      EventBus.emit('pomodoro:break:end', {});
      this.state = 'idle';
    }
  }

  startBreak(minutes, isLong) {
    this.state = 'break';
    this.total = minutes * 60;
    this.remaining = this.total;
    this.startTime = Date.now();
    this.pausedTime = 0;

    this.tick();
    this.intervalId = setInterval(() => this.tick(), 1000);

    EventBus.emit('pomodoro:break', { isLong, duration: minutes });
  }

  createRecord() {
    const record = {
      id: `rec_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`,
      timestamp: Date.now(),
      date: formatDate(new Date()),
      subject: this.subject,
      durationMinutes: this.preset.focus,
      source: 'pomodoro',
      pomodoroSessionId: this.sessionId,
    };

    EventBus.emit('record:created', { record });
    return record;
  }

  stop() {
    clearInterval(this.intervalId);
    this.state = 'idle';
    this.remaining = 0;
    EventBus.emit('pomodoro:stop', {});
  }

  formatTime(seconds) {
    const m = Math.floor(seconds / 60);
    const s = seconds % 60;
    return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
  }
}

export default new PomodoroTimer();
```

```javascript
// search-index.js
class SearchIndex {
  constructor() {
    this.index = new Map(); // keyword → [{ id, type, field, weight }]
  }

  // 构建索引
  build(data) {
    this.index.clear();

    // 学科
    data.subjects.forEach(s => {
      this.addEntry(s.name, s.id, 'subject', 'name', 10);
      this.addEntry(toPinyin(s.name), s.id, 'subject', 'pinyin', 5);
      this.addEntry(toPinyinInitials(s.name), s.id, 'subject', 'initials', 8);
    });

    // 知识点
    Object.values(data.knowledgeState).forEach(k => {
      this.addEntry(k.name, k.id, 'knowledge', 'name', 8);
      this.addEntry(toPinyin(k.name), k.id, 'knowledge', 'pinyin', 4);
    });

    // 学习记录备注
    data.records.forEach(r => {
      if (r.notes) {
        this.addEntry(r.notes, r.id, 'record', 'notes', 3);
      }
    });

    // 功能入口
    const functions = [
      { name: '设置', route: '#/settings', icon: '⚙️' },
      { name: '数据图表', route: '#/data/charts', icon: '📈' },
      { name: '导出数据', route: '#/settings', icon: '📤' },
      // ...
    ];
    functions.forEach(f => {
      this.addEntry(f.name, f.route, 'function', 'name', 6);
    });
  }

  addEntry(text, id, type, field, weight) {
    const tokens = this.tokenize(text);
    tokens.forEach(token => {
      if (!this.index.has(token)) {
        this.index.set(token, []);
      }
      this.index.get(token).push({ id, type, field, weight });
    });
  }

  tokenize(text) {
    // 分词：按空格、标点分割 + n-gram
    const tokens = new Set();
    const clean = text.toLowerCase().replace(/[^\w一-鿿]/g, '');

    // 完整词
    tokens.add(clean);

    // 2-gram
    for (let i = 0; i < clean.length - 1; i++) {
      tokens.add(clean.slice(i, i + 2));
    }

    // 3-gram
    for (let i = 0; i < clean.length - 2; i++) {
      tokens.add(clean.slice(i, i + 3));
    }

    return tokens;
  }

  // 搜索
  search(query, limit = 20) {
    const tokens = this.tokenize(query.toLowerCase());
    const scores = new Map(); // id → { score, type, id }

    tokens.forEach(token => {
      // 精确匹配
      if (this.index.has(token)) {
        this.index.get(token).forEach(entry => {
          const key = `${entry.type}:${entry.id}`;
          const existing = scores.get(key) || { score: 0, ...entry };
          existing.score += entry.weight;
          scores.set(key, existing);
        });
      }

      // 前缀匹配
      for (const [key, entries] of this.index) {
        if (key.startsWith(token) && key !== token) {
          entries.forEach(entry => {
            const k = `${entry.type}:${entry.id}`;
            const existing = scores.get(k) || { score: 0, ...entry };
            existing.score += entry.weight * 0.5; // 前缀匹配权重减半
            scores.set(k, existing);
          });
        }
      }
    });

    return Array.from(scores.values())
      .sort((a, b) => b.score - a.score)
      .slice(0, limit);
  }
}

export default new SearchIndex();
```

```javascript
class TTLCache {
  constructor(defaultTTL = 5 * 60 * 1000) { // 默认 5 分钟
    this.cache = new Map();
    this.defaultTTL = defaultTTL;
  }

  set(key, value, ttl = this.defaultTTL) {

      value,
      expires: Date.now() + ttl,

  }

      return null;
```

```javascript
    caches.open(CACHE_NAME).then((cache) => {
      return cache.addAll(STATIC_ASSETS);
    })

  self.skipWaiting();

// 激活：清理旧缓存
self.addEventListener('activate', (event) => {
  event.waitUntil(
    caches.keys().then((keys) => {
      return Promise.all(
        keys.filter((key) => key !== CACHE_NAME).map((key) => caches.delete(key))
      );
    })
  );
  self.clients.claim();

// 请求拦截：缓存优先策略
self.addEventListener('fetch', (event) => {
  const { request } = event;

  // API 请求：网络优先
  if (request.url.includes('api.github.com')) {
    event.respondWith(
      fetch(request)
        .then((response) => {
          const clone = response.clone();
          caches.open(CACHE_NAME).then((cache) => cache.put(request, clone));
          return response;
        })
        .catch(() => caches.match(request))
    );
    return;
  }

  // 静态资源：缓存优先
  event.respondWith(
    caches.match(request).then((cached) => {
      return cached || fetch(request).then((response) => {
        const clone = response.clone();
        caches.open(CACHE_NAME).then((cache) => cache.put(request, clone));
        return response;
      });
    })
  );
```

```javascript
  const errors = Storage.get('lts_errors') || [];
  errors.push({
    message: error.message,
    stack: error.stack,
    timestamp: Date.now(),
    url: window.location.href,

  Storage.set('lts_errors', errors.slice(-50));
```

### 15.2 网络重试策略

```javascript
async function fetchWithRetry(url, options = {}, maxRetries = 3) {
  for (let i = 0; i < maxRetries; i++) {
    try {
      const response = await fetch(url, options);
      if (response.ok) return response;

      // 429 限流：指数退避
      if (response.status === 429) {
        const delay = Math.pow(2, i) * 1000 + Math.random() * 1000;
        await new Promise((r) => setTimeout(r, delay));
        continue;
      }

      throw new Error(`HTTP ${response.status}`);
    } catch (error) {
      if (i === maxRetries - 1) throw error;
      const delay = Math.pow(2, i) * 1000;
      await new Promise((r) => setTimeout(r, delay));
    }
  }
}
```
